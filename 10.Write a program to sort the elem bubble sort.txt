10.Write a program to sort the elements in an array using Bubble Sort.

#include <iostream>
using namespace std;

// Function to perform Insertion Sort
void insertionSort(float arr[], int n) {
    for (int i = 1; i < n; i++) {
        float key = arr[i];  // Element to be inserted
        int j = i - 1;

        // Move elements of arr[0..i-1], that are greater than key, to one position ahead of their current position
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j = j - 1;
        }

        // Insert the key at the correct position
        arr[j + 1] = key;
    }
}

// Function to print the array
void printArray(float arr[], int n) {
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int main() {
    int n;

    // Input the number of students
    cout << "Enter the number of students: ";
    cin >> n;

    float arr[n];

    // Input the percentages of students
    cout << "Enter the first year percentages of students:\n";
    for (int i = 0; i < n; i++) {
        cout << "Student " << i + 1 << ": ";
        cin >> arr[i];
    }

    // Sort the array using Insertion Sort
    insertionSort(arr, n);

    // Print the sorted array
    cout << "Sorted percentages in ascending order: ";
    printArray(arr, n);

    return 0;
}
