8.Create binary tree. Find height of the tree and print leaf nodes. Find mirror image, print original and mirror image using level-wise printing

#include <iostream>
#include <queue>
using namespace std;

// Definition of a node of the binary tree
class Node {
public:
    int data;
    Node* left;
    Node* right;

    // Constructor to create a new node
    Node(int val) {
        data = val;
        left = right = nullptr;
    }
};

// Class to represent the Binary Tree
class BinaryTree {
public:
    Node* root;

    // Constructor
    BinaryTree() {
        root = nullptr;
    }

    // Function to insert nodes in the binary tree (for simplicity, we'll insert in level-order)
    Node* insertLevelOrder(int arr[], Node* root, int i, int n) {
        // Base case for recursion
        if (i < n) {
            Node* temp = new Node(arr[i]);
            root = temp;

            // Insert left child
            root->left = insertLevelOrder(arr, root->left, 2 * i + 1, n);

            // Insert right child
            root->right = insertLevelOrder(arr, root->right, 2 * i + 1 + 1, n);
        }
        return root;
    }

    // Function to find the height of the tree
    int height(Node* node) {
        if (node == nullptr) {
            return 0;
        }
        int leftHeight = height(node->left);
        int rightHeight = height(node->right);
        return max(leftHeight, rightHeight) + 1;
    }

    // Function to print the leaf nodes
    void printLeafNodes(Node* node) {
        if (node == nullptr) {
            return;
        }
        if (node->left == nullptr && node->right == nullptr) {
            cout << node->data << " ";
        }
        printLeafNodes(node->left);
        printLeafNodes(node->right);
    }

    // Function to print the tree in level-order (Breadth First Search)
    void printLevelOrder(Node* node) {
        if (node == nullptr) {
            return;
        }
        queue<Node*> q;
        q.push(node);

        while (!q.empty()) {
            Node* temp = q.front();
            q.pop();
            cout << temp->data << " ";

            if (temp->left != nullptr) {
                q.push(temp->left);
            }
            if (temp->right != nullptr) {
                q.push(temp->right);
            }
        }
    }

    // Function to create the mirror image of the tree
    Node* mirror(Node* node) {
        if (node == nullptr) {
            return nullptr;
        }
        Node* temp = node->left;
        node->left = mirror(node->right);
        node->right = mirror(temp);
        return node;
    }
};

int main() {
    BinaryTree tree;

    // Example input: an array representing level order insertion into the binary tree
    int arr[] = {1, 2, 3, 4, 5, 6, 7};
    int n = sizeof(arr) / sizeof(arr[0]);
    
    // Insert nodes into the tree
    tree.root = tree.insertLevelOrder(arr, tree.root, 0, n);

    // Print the height of the tree
    cout << "Height of the tree: " << tree.height(tree.root) << endl;

    // Print the leaf nodes
    cout << "Leaf nodes: ";
    tree.printLeafNodes(tree.root);
    cout << endl;

    // Print the original tree in level order
    cout << "Original tree in level order: ";
    tree.printLevelOrder(tree.root);
    cout << endl;

    // Create the mirror image of the tree
    Node* mirrorRoot = tree.mirror(tree.root);

    // Print the mirror tree in level order
    cout << "Mirror tree in level order: ";
    tree.printLevelOrder(mirrorRoot);
    cout << endl;

    return 0;
}
